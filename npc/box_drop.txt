// Random Headgear Box drop: every monster kill has a small chance to drop one item.
// Chance = the "drop_chance" setting in units of 0.01% (5 = 0.05%, the default).
// "drop_box" on: the item is the Random Headgear Box. Off: it is a Headgear Exchange Ticket, which the
// Headgear Exchange NPC (npc/exchange.txt) turns into boxes.
// With "ground_drop" on, the item falls on the ground where the monster died (the player's position if that
// cannot be read) instead of going straight into the inventory. A player whose @autoloot would pick it up still
// gets it directly: the item is on the @autolootitem list, or its drop chance is within the player's @autoloot
// rate and its item type is allowed.

-	script	RHB_BoxDrop	-1,{
	end;

OnInit:
	.chance = callfunc("F_ModSetting", "random-headgear-box", "drop_chance", 5);
	.ground = callfunc("F_ModSetting", "random-headgear-box", "ground_drop", 0);
	.id = callfunc("F_ModSetting", "random-headgear-box", "drop_box", 1) ? 72001 : 72031;
	end;

OnNPCKillEvent:
	if (.chance < 1 || rand(10000) >= .chance) end;

	.@auto = 0;
	if (.ground) {
		if (AUTOLOOT_RATE > 0) {
			.@type = getiteminfo(.id, ITEMINFO_TYPE);
			if (.chance <= AUTOLOOT_RATE && (AUTOLOOT_TYPE == 0 || (AUTOLOOT_TYPE & (1 << .@type)))) .@auto = 1;
		}
		if (!.@auto) {
			.@n = getautolootitems(.@list);
			for (.@u = 0; .@u < .@n; .@u++)
				if (.@list[.@u] == .id) .@auto = 1;
		}
	}

	if (.ground && !.@auto) {
		// Drop it where the monster died. getmapxy cannot look up monsters, so read the dead monster
		// with getunitdata (killedgid). Start from the player's position and only take the monster's
		// if it is a real spot near the player, so a failed lookup falls back to the player.
		getmapxy(.@map$, .@x, .@y, BL_PC);
		getunitdata(killedgid, .@mob);
		.@mx = .@mob[UMOB_X];
		.@my = .@mob[UMOB_Y];
		.@dx = .@mx - .@x;
		.@dy = .@my - .@y;
		if (.@dx < 0) .@dx = -.@dx;
		if (.@dy < 0) .@dy = -.@dy;
		if (.@mx > 0 && .@my > 0 && .@dx <= 30 && .@dy <= 30) {
			.@x = .@mx;
			.@y = .@my;
		}
		makeitem .id, 1, .@map$, .@x, .@y;
	} else {
		getitem .id, 1;
	}
	//dispbottom "A Random Headgear Box dropped!", 0x00FF88;
	end;
}
