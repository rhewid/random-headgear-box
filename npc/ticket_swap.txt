// Ticket Swap -- trades Headgear Exchange Tickets (72031) and Costume Exchange Tickets (72032) 1:1, in both directions.

prontera,188,193,4	script	Ticket Swap#rhb	4_M_SAGE_A,{
	mes "[Ticket Swap]";
	mes "I trade Headgear Exchange Tickets and Costume Exchange Tickets one for one, either way.";
	mes "Headgear tickets: ^0055FF" + countitem(72031) + "^000000";
	mes "Costume tickets: ^0055FF" + countitem(72032) + "^000000";
	next;
	.@c = select("Headgear tickets -> Costume tickets:Costume tickets -> Headgear tickets:Cancel");
	if (.@c == 3) close;
	if (.@c == 1) {
		.@from = 72031;
		.@to = 72032;
	} else {
		.@from = 72032;
		.@to = 72031;
	}
	.@have = countitem(.@from);
	if (.@have < 1) {
		mes "[Ticket Swap]";
		mes "You have no " + getitemname(.@from) + "s.";
		close;
	}
	mes "[Ticket Swap]";
	mes "How many " + getitemname(.@from) + "s do you want to swap for " + getitemname(.@to) + "s? (1 to " + .@have + ")";
	next;
	input .@n, 0, .@have;
	if (.@n < 1) close;
	if (countitem(.@from) < .@n) {
		mes "[Ticket Swap]";
		mes "You do not have that many.";
		close;
	}
	if (!checkweight(.@to, .@n)) {
		mes "[Ticket Swap]";
		mes "You are carrying too much. Make some room first!";
		close;
	}
	delitem .@from, .@n;
	getitem .@to, .@n;
	mes "[Ticket Swap]";
	mes "Done: " + .@n + " x ^008800" + getitemname(.@to) + "^000000.";
	close;
}
