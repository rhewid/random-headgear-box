// Costume Tailor -- turns a headgear into a costume that looks the same. (The same NPC as in the
// headgear-to-costume mod; do not run both mods at once.)
//
// The costume twin of every headgear is a normal item in the id block 73000-75999 (see README.md and the
// era folder's db/item_db.yml). Which headgear maps to which twin is in the era folder's npc/htc_table.txt:
// .src[] is the sorted list of headgear ids and the twin of .src[i] is .base + i. This script finds the
// headgear with a binary search, so it works for any size of table and needs no change per era.
//
// What the player gets: the costume. Cards in the headgear go back to the player. Refine, enchantments and
// random options are lost. Fee: "costume_fee" setting (zeny).

prontera,181,190,4	script	Costume Tailor#rhb	4_F_02,{
	mes "[Costume Tailor]";
	mes "Hand me a headgear and I will sew a costume that looks exactly like it.";
	mes "Cards in it come back to you. ^FF0000Refine, enchantments and bonus options are lost.^000000";
	mes "Fee: ^0055FF" + callfunc("F_InsertComma", .fee) + " Zeny^000000.";
	next;
	if (select("Make a costume:Cancel") == 2) close;

	getinventorylist;
	.@cnt = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_equip[.@i]) continue;
		.@cid = callsub(L_Find, @inventorylist_id[.@i]);
		if (.@cid <= 0) continue;
		.@sel_arr[.@cnt] = .@i;
		.@sel_idx[.@cnt] = @inventorylist_idx[.@i];
		.@sel_id[.@cnt] = @inventorylist_id[.@i];
		.@sel_cos[.@cnt] = .@cid;
		.@cnt++;
	}
	if (.@cnt == 0) {
		mes "[Costume Tailor]";
		mes "You are not carrying a headgear I can work with. Unequip it first.";
		close;
	}

	.@page = 0;
	.@per = 30;
	while (1) {
		.@start = .@page * .@per;
		.@end = .@start + .@per;
		if (.@end > .@cnt) .@end = .@cnt;
		.@menu$ = "";
		for (.@j = .@start; .@j < .@end; .@j++) {
			.@r = @inventorylist_refine[.@sel_arr[.@j]];
			.@menu$ += (.@r > 0 ? "+" + .@r + " " : "") + getitemname(.@sel_id[.@j]) + ":";
		}
		.@shown = .@end - .@start;
		.@has_next = (.@end < .@cnt);
		.@has_prev = (.@page > 0);
		if (.@has_next) .@menu$ += "Next page:";
		if (.@has_prev) .@menu$ += "Previous page:";
		.@menu$ += "Cancel";
		.@c = select(.@menu$);
		if (.@c <= .@shown) { .@pick = .@start + .@c - 1; break; }
		.@c -= .@shown;
		if (.@has_next) {
			if (.@c == 1) { .@page++; continue; }
			.@c--;
		}
		if (.@has_prev) {
			if (.@c == 1) { .@page--; continue; }
		}
		close;
	}

	.@arr = .@sel_arr[.@pick];
	.@idx = .@sel_idx[.@pick];
	.@id = .@sel_id[.@pick];
	.@cos = .@sel_cos[.@pick];
	.@ref = @inventorylist_refine[.@arr];
	.@ncards = 0;
	for (.@k = 1; .@k <= 4; .@k++) {
		.@cd = getd("@inventorylist_card" + .@k + "[" + .@arr + "]");
		if (.@cd > 0 && getiteminfo(.@cd, ITEMINFO_TYPE) == 6) {
			.@card[.@ncards] = .@cd;
			.@ncards++;
		}
	}

	mes "[Costume Tailor]";
	mes "Make a costume from ^0000FF" + (.@ref > 0 ? "+" + .@ref + " " : "") + getitemname(.@id) + "^000000?";
	mes "You get: ^008800" + getitemname(.@cos) + "^000000.";
	if (.@ncards) {
		mes "Cards returned:";
		for (.@k = 0; .@k < .@ncards; .@k++) mes " - " + getitemname(.@card[.@k]);
	}
	mes "^FF0000The headgear is used up. Refine, enchantments and bonus options are lost.^000000";
	mes "Fee: ^0055FF" + callfunc("F_InsertComma", .fee) + " Zeny^000000.";
	next;
	if (select("Yes, make the costume:Cancel") == 2) close;

	if (Zeny < .fee) {
		mes "[Costume Tailor]";
		mes "You need " + callfunc("F_InsertComma", .fee) + " Zeny.";
		close;
	}
	// The inventory may have changed while the menu was open: find the item again by inventory index.
	getinventorylist;
	.@ok = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_idx[.@i] == .@idx && @inventorylist_id[.@i] == .@id && !@inventorylist_equip[.@i]) {
			.@ok = 1;
			break;
		}
	}
	if (!.@ok) {
		mes "[Costume Tailor]";
		mes "That headgear is no longer where it was. Please try again.";
		close;
	}
	if (!checkweight(.@cos, 1)) {
		mes "[Costume Tailor]";
		mes "You are carrying too much. Make some room first!";
		close;
	}

	delitemidx .@idx, 1;
	if (.fee) Zeny -= .fee;
	for (.@k = 0; .@k < .@ncards; .@k++) getitem .@card[.@k], 1;
	getitem .@cos, 1;
	specialeffect2 EF_COIN;
	mes "[Costume Tailor]";
	mes "Here you go: ^008800" + getitemname(.@cos) + "^000000.";
	close;

// Binary search of a headgear id in the era's table (HTC_Table). Returns the costume item id, or 0.
L_Find:
	.@x = getarg(0);
	.@lo = 0;
	.@hi = getvariableofnpc(.count, "HTC_Table") - 1;
	while (.@lo <= .@hi) {
		.@m = (.@lo + .@hi) / 2;
		.@v = getvariableofnpc(.src[.@m], "HTC_Table");
		if (.@v == .@x) return getvariableofnpc(.base, "HTC_Table") + .@m;
		if (.@v < .@x) .@lo = .@m + 1; else .@hi = .@m - 1;
	}
	return 0;

OnInit:
	.fee = callfunc("F_ModSetting", "random-headgear-box", "costume_fee", 10000);
	end;
}
